firebase.initializeApp(firebaseConfig);

const database = firebase.database();

const studentsRef = database.ref('students');

const barcodeInput = document.getElementById('barcode-input');

const scanMessage = document.getElementById('scan-message');

let localStudents = {};

document.addEventListener('click', (e) => {

if(e.target.tagName !== 'INPUT' && e.target.tagName !== 'BUTTON') {

barcodeInput.focus();

}

});

barcodeInput.addEventListener('keypress', function (e) {

if (e.key === 'Enter') {

const code = barcodeInput.value.trim();

if (code) {

registerAttendance(code);

barcodeInput.value = '';

}

}

});

function registerAttendance(code) {

if (localStudents[code]) {

const student = localStudents[code];

const currentCount = student.attendanceCount || 0;

const today = new Date().toLocaleDateString('ar-EG');

const dates = student.attendanceDates || [];

dates.push(today);

studentsRef.child(code).update({

attendanceCount: currentCount + 1,

attendanceDates: dates

});

scanMessage.style.color = "green";

scanMessage.innerText = `تم تسجيل حضور: ${student.name} بنجاح!`;

} else {

scanMessage.style.color = "red";

scanMessage.innerText = `الكود (${code}) غير موجود بالسيستم!`;

}

setTimeout(() => { scanMessage.innerText = ''; }, 3000);

}

document.getElementById('student-form').addEventListener('submit', function (e) {

e.preventDefault();

const code = document.getElementById('student-code').value.trim();

const name = document.getElementById('student-name').value.trim();

const phone = document.getElementById('student-phone').value.trim();

const parentPhone = document.getElementById('parent-phone').value.trim();

const fee = parseFloat(document.getElementById('monthly-fee').value);

const existing = localStudents[code] || {};

studentsRef.child(code).set({

code: code,

name: name,

phone: phone,

parentPhone: parentPhone,

fee: fee,

isPaid: existing.isPaid || false,

examScore: existing.examScore || 'لم تحدد',

attendanceCount: existing.attendanceCount || 0,

attendanceDates: existing.attendanceDates || []

});

this.reset();

});

studentsRef.on('value', (snapshot) => {

localStudents = snapshot.val() || {};

renderTable(localStudents);

});

function togglePaid(code) {

const currentPaid = localStudents[code].isPaid;

studentsRef.child(code).update({ isPaid: !currentPaid });

}

function updateScore(code, score) {

studentsRef.child(code).update({ examScore: score });

}

function sendWhatsApp(code) {

const s = localStudents[code];

const paidStatus = s.isPaid ? "تم السداد ✅" : "لم يتم السداد ❌";

const studentPageUrl = window.location.href.replace('index.html', 'student.html') + `?code=${s.code}`;

const message = `تقرير الطالب: ${s.name}%0A` +

`الكود: ${s.code}%0A` +

`عدد حصص الحضور: ${s.attendanceCount || 0}%0A` +

`درجة الامتحان: ${s.examScore}%0A` +

`حالة المصاريف: ${paidStatus}%0A` +

`رابط متابعة الطالب مباشرة:%0A${studentPageUrl}`;

window.open(`https://wa.me/${s.parentPhone}?text=${message}`, '_blank');

}

function copyLink(code) {

const url = window.location.href.replace('index.html', 'student.html') + `?code=${code}`;

navigator.clipboard.writeText(url);

alert('تم نسخ رابط صفحة الطالب بنجاح!');

}

function renderTable(studentsData) {

const tbody = document.getElementById('students-table-body');

tbody.innerHTML = '';

let count = 0;

let revenue = 0;

for (let code in studentsData) {

count++;

const s = studentsData[code];

if (s.isPaid) revenue += (s.fee || 0);

const tr = document.createElement('tr');

tr.innerHTML = `

<td>${s.code}</td>

<td>${s.name}</td>

<td>${s.parentPhone}</td>

<td><input type="text" value="${s.examScore}" style="width:70px;" onchange="updateScore('${s.code}', this.value)"></td>

<td>

<button style="background:${s.isPaid ? '#28a745' : '#dc3545'}" onclick="togglePaid('${s.code}')">

${s.isPaid ? 'تم السداد' : 'لم يتم'}

</button>

</td>

<td>${s.attendanceCount || 0}</td>

<td>

<button class="btn-whatsapp" onclick="sendWhatsApp('${s.code}')">واتساب 📱</button>

<button class="btn-link" onclick="copyLink('${s.code}')">رابط الطالب 🔗</button>

</td>

`;

tbody.appendChild(tr);

}

document.getElementById('total-students').innerText = count;

document.getElementById('total-revenue').innerText = revenue;

}
